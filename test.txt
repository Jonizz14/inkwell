(function() {
    // ==========================================
    // 1. FOCUS & TAB BYPASS (Fonda jimgina ishlaydi)
    // ==========================================
    const lockProperty = (proto, prop, descriptor) => {
        try {
            Object.defineProperty(proto, prop, descriptor);
        } catch (e) {}
    };

    lockProperty(Document.prototype, 'hidden', { get: () => false, configurable: true });
    lockProperty(Document.prototype, 'visibilityState', { get: () => 'visible', configurable: true });
    lockProperty(Document.prototype, 'webkitHidden', { get: () => false, configurable: true });
    lockProperty(Document.prototype, 'webkitVisibilityState', { get: () => 'visible', configurable: true });
    
    Document.prototype.hasFocus = () => true;
    window.hasFocus = () => true;

    const blockEvent = (e) => {
        e.stopImmediatePropagation();
    };

    const spyEvents = ['blur', 'mouseleave', 'visibilitychange', 'webkitvisibilitychange', 'focusout', 'pagehide'];
    spyEvents.forEach(evt => {
        window.addEventListener(evt, blockEvent, { capture: true, passive: true });
        document.addEventListener(evt, blockEvent, { capture: true, passive: true });
    });

    // ==========================================
    // 2. FULLSCREEN KILLER
    // ==========================================
    const killFS = () => Promise.resolve();
    ['requestFullscreen', 'webkitRequestFullscreen', 'mozRequestFullScreen', 'msRequestFullscreen'].forEach(fn => {
        if (Element.prototype[fn]) Element.prototype[fn] = killFS;
    });

    lockProperty(document, 'fullscreenElement', { get: () => null, configurable: true });
    lockProperty(document, 'webkitFullscreenElement', { get: () => null, configurable: true });
    lockProperty(document, 'fullscreenEnabled', { get: () => { return true; }, configurable: true });

    // ==========================================
    // 3. GLOBAL TEST SCANNER & JUMP SYSTEM
    // ==========================================
    window.ninjaScanAllSections = function() {
        const questions = document.querySelectorAll('.question, .qtext, [class*="question"], div[id*="question"], .test-item, .quiz-question');
        
        console.info(`[SYS] Global cache dump: Found ${questions.length} items across blocks.`);

        if (questions.length === 0) {
            console.warn("[WARN] No standard blocks found. Dumping main content container...");
            console.log(document.body.innerText);
            return;
        }

        questions.forEach((q, index) => {
            const qText = q.innerText.trim();
            console.log(`\n-----------------------------------------`);
            console.log(`[SAVOL #${index + 1}] ->`, qText);
            
            const answers = q.querySelectorAll('input[type="radio"] ~ *, input[type="checkbox"] ~ *, .answer label, [class*="answer"], .option');
            if (answers.length > 0) {
                answers.forEach((ans, aIdx) => {
                    console.log(`   (opt_${aIdx + 1}): ${ans.innerText.trim()}`);
                });
            }
        });
        console.log(`-----------------------------------------`);
        console.info("[SYS] Jump tip: Use jumpTo(number) to navigate.");
    };

    // Istalgan savol raqamiga sakrash funksiyasi (Masalan: jumpTo(3))
    window.jumpTo = function(questionNumber) {
        const questions = document.querySelectorAll('.question, .qtext, [class*="question"], div[id*="question"], .test-item, .quiz-question');
        const index = questionNumber - 1;
        
        if (questions && questions[index]) {
            questions[index].scrollIntoView({ behavior: 'smooth', block: 'center' });
            console.info(`[SYS] Successfully jumped to Question #${questionNumber}`);
        } else {
            // Agar savollar alohida sahifalarga bo'lingan bo'lsa, navigation tugmalarini qidiramiz
            const navButtons = document.querySelectorAll('.pagination a, .quiz-navigation button, [class*="page"], [class*="nav"]');
            if (navButtons && navButtons[index]) {
                navButtons[index].click();
                console.info(`[SYS] Clicked navigation for Question #${questionNumber}`);
            } else {
                console.warn(`[WARN] Question #${questionNumber} not found on this DOM tree.`);
            }
        }
    };

    // ==========================================
    // 4. HOTKEYS (Qaynoq tugmalar)
    // ==========================================
    window.addEventListener('keydown', (e) => {
        // Ctrl + Shift + L -> Konsolni tozalaydi
        if (e.ctrlKey && e.shiftKey && e.key === 'L') {
            console.clear();
        }
        // Ctrl + Shift + F -> Barcha savollarni va javoblarni chiqaradi
        if (e.ctrlKey && e.shiftKey && e.key === 'F') {
            e.preventDefault();
            window.ninjaScanAllSections();
        }
    });

    console.log("[DOM] Initialized state: status 200 OK.");
})();
