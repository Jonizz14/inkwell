(function() {
    // ==========================================
    // 1. FOCUS & TAB BYPASS (Fonda jimgina ishlaydi)
    // ==========================================
    const lockProperty = (proto, prop, descriptor) => {
        try { Object.defineProperty(proto, prop, descriptor); } catch (e) {}
    };

    lockProperty(Document.prototype, 'hidden', { get: () => false, configurable: true });
    lockProperty(Document.prototype, 'visibilityState', { get: () => 'visible', configurable: true });
    lockProperty(Document.prototype, 'webkitHidden', { get: () => false, configurable: true });
    lockProperty(Document.prototype, 'webkitVisibilityState', { get: () => 'visible', configurable: true });
    
    Document.prototype.hasFocus = () => true;
    window.hasFocus = () => true;

    const blockEvent = (e) => { e.stopImmediatePropagation(); };
    ['blur', 'mouseleave', 'visibilitychange', 'webkitvisibilitychange', 'focusout', 'pagehide'].forEach(evt => {
        window.addEventListener(evt, blockEvent, { capture: true, passive: true });
        document.addEventListener(evt, blockEvent, { capture: true, passive: true });
    });

    // ==========================================
    // 2. FULLSCREEN KILLER
    // ==========================================
    const killFS = () => Promise.resolve();
    ['requestFullscreen', 'webkitRequestFullscreen', 'mozRequestFullScreen', 'msRequestFullscreen'].forEach(fn => {
        if (Element.prototype[fn]) Element.prototype[fn] = killFS;
    });

    lockProperty(document, 'fullscreenElement', { get: () => null, configurable: true });
    lockProperty(document, 'webkitFullscreenElement', { get: () => null, configurable: true });
    lockProperty(document, 'fullscreenEnabled', { get: () => true, configurable: true });

    // ==========================================
    // 3. SMART ANSWER FINDER (To'g'ri javoblarni qidiruvchi)
    // ==========================================
    window.findCorrectAnswers = function() {
        const questions = document.querySelectorAll('.question, .qtext, [class*="question"], div[id*="question"], .test-item, .quiz-question');
        
        if (questions.length === 0) {
            console.warn("[WARN] Standart savol bloklari topilmadi. Sahifa ma'lumotlari tahlil qilinmoqda...");
            return;
        }

        console.info(`[SYS] Topildi: ${questions.length} ta savol. To'g'ri javoblar qidirilmoqda...`);

        questions.forEach((q, index) => {
            const qText = q.innerText.trim();
            console.log(`\n=========================================`);
            console.log(`[SAVOL #${index + 1}]: ${qText}`);

            const answers = q.querySelectorAll('input[type="radio"], input[type="checkbox"], .answer label, [class*="answer"], [class*="option"], label');

            if (answers.length > 0) {
                answers.forEach((ans, aIdx) => {
                    const ansText = ans.innerText || ans.value || `Variant ${aIdx + 1}`;
                    let mark = "   [-] ";

                    if (ans.getAttribute('data-correct') === 'true' || ans.classList.contains('correct') || ans.classList.contains('right')) {
                        mark = "   [✅ TO'G'RI JAVOB]: ";
                        console.log(`%c${mark}${ansText.trim()}`, "color: #00ff00; font-weight: bold;");
                    } else {
                        console.log(`${mark}${ansText.trim()}`);
                    }
                });
            }
        });
        console.log(`=========================================`);
        console.info("[SYS] Qidiruv yakunlandi. Serverga hech qanday ma'lumot yuborilmadi.");
    };

    // ==========================================
    // 4. JUMP SYSTEM (Istalgan savolga o'tish)
    // ==========================================
    window.jumpTo = function(questionNumber) {
        const allElements = document.querySelectorAll('.question, .qtext, [class*="question"], div[id*="question"], .test-item, .quiz-question');
        const index = questionNumber - 1;
        
        if (allElements && allElements[index]) {
            allElements[index].scrollIntoView({ behavior: 'smooth', block: 'center' });
            console.info(`[SYS] Sakrlandi: Savol #${questionNumber}`);
        } else {
            const navButtons = document.querySelectorAll('.pagination a, .quiz-navigation button, [class*="page"], [class*="nav"]');
            if (navButtons && navButtons[index]) {
                navButtons[index].click();
                console.info(`[SYS] Sahifa tugmasi bosildi: #${questionNumber}`);
            } else {
                console.warn(`[WARN] #${questionNumber} - topilmadi.`);
            }
        }
    };

    // ==========================================
    // 5. HOTKEYS
    // ==========================================
    window.addEventListener('keydown', (e) => {
        if (e.ctrlKey && e.shiftKey && e.key === 'L') {
            console.clear();
        }
        // Ctrl + Shift + S -> To'g'ri javoblarni qidirib chiqaradi
        if (e.ctrlKey && e.shiftKey && e.key === 'S') {
            e.preventDefault();
            window.findCorrectAnswers();
        }
    });

    console.log("[DOM] Initialized state: status 200 OK.");
})();
